number[] function bubble_sort(number[] arr):
	number n = 5
	for number i = 0, i < n - 1, 1:
		for number j = 0, j < n - i - 1, 1:
			if arr[j] > arr[j+1]:
				number temp = arr[j]
				arr[j] = arr[j+1]
				arr[j+1] = temp
	return arr
number[] example = [1,10,69,5,3]
example = bubble_sort(example)
